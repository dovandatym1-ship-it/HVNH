// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract TechcombankLoyaltyToken is ERC20, Ownable {
    // Thưởng điểm khi chi tiêu thẻ
    function rewardPoints(address customer, uint256 points, string calldata txRef) 
        external onlyOwner {
        _transfer(owner(), customer, points * 10 ** decimals());
    }

    // Đổi voucher và đốt token tránh lạm phát
    function redeemVoucher(uint256 cost, string calldata partner, string calldata code) 
        external returns (bytes32) {
        _burn(msg.sender, cost * 10 ** decimals());
        // Emit Event đối soát tức thì
    }
}
